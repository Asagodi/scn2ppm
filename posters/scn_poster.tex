\column{1.0}
\block{\bt{1}{Setup \& The Projective Map}}{
\begin{minipage}[t]{0.6\linewidth}\vspace{0pt}\normalsize
$M$ neurons encode $\xx\in\reals^N$ via readout $\yy = \vv{D}\vv{r}$~\ref{ref:boerlin}, \ref{ref:calaim}:
decay $\dot\yy = -\lambda\yy$; spike $\yy \mapsto \yy + \DD_i$ at $\DD_i^\top\yy = \alpha_i$.

\vspace{1mm}
Each spike bound $\Bsp_i \subset \reals^N$ is an $(N{-}1)$-dim hyperplane. On $\Bsp_i$: $\yy = \alpha_i\DD_i + \vv{U}_i\xxi$, where $\xxi\in\reals^{N-1}$ is the tangent coordinate. The full dynamics is a \textbf{piecewise projective map} on the disjoint union $\bigsqcup_i \Bsp_i$. %: a point on $\Bsp_i$ is mapped to the next firing bound $\Bsp_j$ via the formula below.
\end{minipage}\hfill
\begin{minipage}[t]{0.38\linewidth}\vspace{0pt}\normalsize
\colorbox{accentsoft}{\parbox{0.96\linewidth}{%
  \vspace{1.5mm}\centering\normalsize
  $\displaystyle\xxi_j = \alpha_j\,\frac{(\alpha_i{+}1)\vv{w}_{ji}+\vv{R}_{ji}\,\xxi_i}{(\alpha_i{+}1)\cos\psi_{ji}+\vv{v}_{ji}^\top\xxi_i}$
  \quad
  $\vv{M}_{j\leftarrow i} {=} \begin{pmatrix} \alpha_j\vv{R}_{ji} & \alpha_j(\alpha_i{+}1)\vv{w}_{ji} \\ \vv{v}_{ji}^\top & (\alpha_i{+}1)\cos\psi_{ji} \end{pmatrix}$
  \vspace{1.5mm}
}}
\end{minipage}

\vspace{3mm}
{\small\textbf{\color{navy}Inter-bound geometry.}\;
 With $\vv{U}_i$ the tangent frame of $\Bsp_i$ (its $N{-}1$ columns span the hyperplane):
 $\cos\psi_{ji}=\DD_j^\top\DD_i$ (decoder cosine);\;
 $\vv{v}_{ji}=\vv{U}_i^\top\DD_j$, $\;\vv{w}_{ji}=\vv{U}_j^\top\DD_i$ (decoders projected into the tangent frames);\;
 $\vv{R}_{ji}=\vv{U}_j^\top\vv{U}_i$ (maps tangent frame $i\!\to\!j$).\par}

\begin{columns}
\column{1.0}
\block{\bt{2}{Orbits = Eigenvectors of the Return Map}}{
\begin{minipage}[t]{\linewidth}\normalsize
For a firing word $\sigma=(b_0,\ldots,b_{L-1})$, composing the per-spike matrices $\vv{M}_{b_{k+1}\leftarrow b_k}$ gives the \textbf{symbolic return map}
$\vv{M}_\sigma = \prod_{k} \vv{M}_{b_{k+1}\leftarrow b_k} \in \reals^{N\times N}$, which propagates the tangent coordinate $\xxi\in\Bsp_{b_0}$ once around the cycle.\quad
\textbf{Periodic orbit $\leftrightarrow$ eigenvector.} Picture the homogeneous coordinate $\tilde\xxi=(\xxi^{*\top},1)^\top$ as a ray through the origin of $\reals^N$, and $\xxi^*$ as the unique point where this ray pierces the affine slice $\{x_N{=}1\}$. A period-$L$ cycle returns the ray to itself, $\vv{M}_\sigma\tilde\xxi^*=\mu_1\tilde\xxi^*$: $\xxi^*$ is the dehomogenization of the dominant eigenvector. The scale $\mu_1$ does not move the pierce point and cancels.\par
\textbf{Stability spectrum.} The return map $F=\mathrm{dh}\circ\vv{M}_\sigma$ on $\Bsp_{b_0}$ has, at its fixed point $\xxi^*$, a Jacobian with $N{-}1$ eigenvalues. Numerator $\mu_k$ along a transverse eigenvector $\vv{v}_k$, denominator $\mu_1$ along the fixed ray: the displacement on the slice is multiplied by the ratio $\mu_k/\mu_1$, one per independent perturbation direction. The orbit attracts iff every multiplier is inside the unit disk: $|\mu_k/\mu_1|<1$ for $k\ge 2$.

\vspace{1mm}
\btnonum{Cobweb.}\;
First-return map $F:\Bsp_1\to\Bsp_1$ has two M\"obius branches (self \& cross) joined at the partition point $\xi_{\mathrm{part}}$. Period-$L$ orbit visits $\Bsp_1$ a subset of times $\Rightarrow$ cobweb cycle.

\vspace{1mm}
\btnonum{Devil's staircase.}\;
Mode-locked plateaus at every rational firing ratio $\rho$~\ref{ref:arnold},\,\ref{ref:herman}. Period-$1\to 2$ boundary: $\alpha_j = \alpha_i\cos\psi_{ji}$.
\end{minipage}
}
\end{columns}

\column{0.45}
\block{\bt{6}{Inverse Problem}}{
\begin{minipage}[t]{\linewidth}\normalsize
\textbf{Given} a firing word $\sigma$, \textbf{find} drives $\bm{\alpha}$ that realize it.

\vspace{1mm}
\textbf{Admissibility margin} (worst-case target $-$ competitor gap):
\[
  m(\bm{\alpha}) = \min_{k}\;\bigl(
    c_{b_{k+1},k} - \max_{j\neq b_{k+1}} c_{j,k}
  \bigr)
\]
$m{>}0$ realizes the word; maximize $m$ for robustness.

\vspace{1mm}

  \textbf{\color{accent}Working memory.}\;
  Margin shrinks with word length $\Rightarrow$ matches the behavioral $4{\pm}1$ limit~\ref{ref:cowan}.
  Same $(\vv{D},\bm{\alpha})$ stores multiple words (\emph{multistability}); varying $\xx$ navigates the staircase to switch between stored items.
  \emph{Storage is closed-form}: the drives $\bm{\alpha}$ are read off the linear program.
  \emph{Retrieval is autonomous}: each stored word is the leading eigenvector of its own return map, so the orbit unfolds from any initial state in the basin.%
